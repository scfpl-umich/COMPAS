[Sod-5Eq]    BUILD=5Eq     inputs=prob/inputs
# The base inputs run the shock tube along x with outflow boundaries. dir-y and dir-z run it
# along y and z, and walls closes it with slip walls in x, symmetry in y and no-slip walls in z.
# water-air is the two-material shock tube with stiffened-gas water (rho 1000, P 1e9, gamma 4.4,
# pinf 6e8) left of a diaphragm at 0.7 and air (rho 50, P 1e5, gamma 1.4) to the right, in SI units.

dir-y   max_step=20 prob.direction=1 geometry.prob_hi=0.1 1.0 0.1 amr.n_cell=4 256 4 amr.blocking_factor_x=4 amr.blocking_factor_y=16
dir-z   max_step=20 prob.direction=2 geometry.prob_hi=0.1 0.1 1.0 amr.n_cell=4 4 256 amr.blocking_factor_x=4 amr.blocking_factor_z=16
walls   max_step=20 run.lo_bc=4 3 5 run.hi_bc=4 3 5
water-air max_step=20 prob.rho_L=1000 prob.P_L=1e9 prob.rho_R=50 prob.P_R=1e5 prob.mat_R=2 prob.x_diaphragm=0.7 prob.EOS.gamma_1=4.4 prob.EOS.pinf_1=6e8 prob.EOS.gamma_2=1.4 prob.EOS.pinf_2=0.0
